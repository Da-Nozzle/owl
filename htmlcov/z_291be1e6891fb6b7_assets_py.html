139 def dbt_step(context: dg.AssetExecutionContext, dbt: DbtCliResource) -> Iterator[object]: 

140 run_labels = label_run(context, labels) 

141 # An unpartitioned step in a partitioned run still sees the run's partition key. 

142 snapshot = context.partition_key if partitions else None 

143 with environment_variable(SNAPSHOT_VARIABLE, snapshot) if snapshot else nullcontext(): 

144 build = dbt.cli(["build"], context=context) 

145 events = ( 

146 build.stream() 

147 .fetch_column_metadata(with_column_lineage=column_lineage) 

148 .fetch_row_counts() 

149 ) 

150 for event in events: 

151 counted = snapshot_metadata(event, build, snapshot) if snapshot else event 

152 yield with_labels(with_compiled_sql(counted, build), run_labels) 

153 keep_main_state(build.target_path / "manifest.json", main_state_dir) 

154 

155 return dbt_step

158def source_layers(manifest: Mapping[str, Any]) -> dict[str, set[str]]: 

159 """Each source's layers that hold a model of it, by source folder name, sorted by source.""" 

160 layers: dict[str, set[str]] = {} 

161 for node in manifest["nodes"].values(): 

162 fqn = node["fqn"] # [project, layer, folder..., model] 

163 if node["resource_type"] != "model" or fqn[1] not in SOURCE_LAYERS: 

164 continue 

165 if len(fqn) < 4: 

166 raise ValueError( 

167 f"{node['original_file_path']} is not in a source folder: " 

168 f"put it in models/{fqn[1]}/<source>/" 

169 ) 

170 layers.setdefault(fqn[2], set()).add(fqn[1]) 

171 return dict(sorted(layers.items()))
